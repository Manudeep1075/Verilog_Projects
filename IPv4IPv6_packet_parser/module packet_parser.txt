module packet_parser (
    input clk,
    input rst_n,

    // Input stream (one byte per clock when rx_valid=1)
    input [7:0]  rx_data,
    input rx_valid,
    input rx_last,
    output rx_ready,

    // ── IPv4 parsed fields ──────────────────────────────────
    output reg ipv4_valid,
    output reg  [3:0]  ipv4_version,
    output reg  [3:0]  ipv4_ihl,
    output reg  [7:0]  ipv4_dscp_ecn,
    output reg  [15:0] ipv4_total_len,
    output reg  [15:0] ipv4_id,
    output reg  [2:0]  ipv4_flags,
    output reg  [12:0] ipv4_frag_off,
    output reg  [7:0]  ipv4_ttl,
    output reg  [7:0]  ipv4_protocol,
    output reg  [15:0] ipv4_checksum,
    output reg  [31:0] ipv4_src_addr,
    output reg  [31:0] ipv4_dst_addr,

    // ── IPv6 parsed fields ──────────────────────────────────
    output reg ipv6_valid,
    output reg  [3:0]  ipv6_version,
    output reg  [7:0]  ipv6_traffic_class,
    output reg  [19:0] ipv6_flow_label,
    output reg  [15:0] ipv6_payload_len,
    output reg  [7:0]  ipv6_next_header,
    output reg  [7:0]  ipv6_hop_limit,
    output reg  [127:0] ipv6_src_addr,
    output reg  [127:0] ipv6_dst_addr,

    // ── Status ───────────────────────────────────────────────
    output reg parse_done,
    output reg  [1:0]  ip_version_out,
    output reg parse_error
);

assign rx_ready = 1'b1;

// ── Global byte counter — resets after rx_last ──────────────
reg [7:0] gcnt;

always @(posedge clk or negedge rst_n)
  begin
    if (!rst_n)
        gcnt <= 8'd0;
    else if (rx_valid)
      begin
        if (rx_last)
          gcnt <= 8'd0;
        else
          gcnt <= gcnt + 8'd1;
    end
end

// ── Latch EtherType high byte (byte 12) ─────────────────────
reg [7:0] etype_hi;
always @(posedge clk or negedge rst_n)
  begin
    if (!rst_n)
      etype_hi <= 8'd0;
    else if (rx_valid && gcnt == 8'd12)
      etype_hi <= rx_data;
end

// ── Latch packet type on byte 13 (low byte of EtherType) ────
reg pkt_is_ipv4, pkt_is_ipv6;
always @(posedge clk or negedge rst_n)
  begin
    if (!rst_n)
      begin
        pkt_is_ipv4 <= 1'b0;
        pkt_is_ipv6 <= 1'b0;
      end
    else if (rx_valid && gcnt == 8'd13)
      begin
        pkt_is_ipv4 <= (etype_hi == 8'h08) && (rx_data == 8'h00);
        pkt_is_ipv6 <= (etype_hi == 8'h86) && (rx_data == 8'hDD);
      end 
    else if (rx_valid && rx_last)
      begin
        pkt_is_ipv4 <= 1'b0;
        pkt_is_ipv6 <= 1'b0;
      end
  end
// Offset from start of IP header (Ethernet header is 14 bytes)
wire [7:0] ip_off = gcnt - 8'd14;

// ── Main parse block ─────────────────────────────────────────
always @(posedge clk or negedge rst_n) begin
    if (!rst_n)
      begin
        ipv4_valid <= 0;
        ipv6_valid  <= 0;
        parse_done <= 0;
        parse_error <= 0;
        ip_version_out <= 0;
        ipv4_version <= 0;
        ipv4_ihl <= 0;
        ipv4_dscp_ecn <= 0;
        ipv4_total_len <= 0;
        ipv4_id <= 0;
        ipv4_flags <= 0;
        ipv4_frag_off  <= 0;
        ipv4_ttl <= 0;
        ipv4_protocol <= 0;
        ipv4_checksum  <= 0;
        ipv4_src_addr <= 0;
        ipv4_dst_addr <= 0;
        ipv6_version   <= 0;
        ipv6_traffic_class <= 0;
        ipv6_flow_label <= 0;
        ipv6_payload_len  <= 0;
        ipv6_next_header <= 0;
        ipv6_hop_limit   <= 0;
        ipv6_src_addr  <= 0;
        ipv6_dst_addr <= 0;
    end else begin
        // Pulse signals — cleared every cycle unless re-asserted
        parse_done  <= 1'b0;
        parse_error <= 1'b0;

        // Clear valid outputs at the very start of each new packet
        if (rx_valid && gcnt == 8'd0) begin
            ipv4_valid <= 1'b0;
            ipv6_valid <= 1'b0;
            ip_version_out <= 2'd0;
        end

        if (rx_valid && gcnt >= 8'd14) begin

            // ════════════════════════════════════════════════
            //  IPv4 — 20-byte fixed header
            // ════════════════════════════════════════════════
            if (pkt_is_ipv4) begin
                case (ip_off)
                    8'd0:  begin
                               ipv4_version <= rx_data[7:4];
                               ipv4_ihl     <= rx_data[3:0];
                               if (rx_data[7:4] != 4'd4)
                                   parse_error <= 1'b1;
                           end
                    8'd1:  ipv4_dscp_ecn       <= rx_data;
                    8'd2:  ipv4_total_len[15:8] <= rx_data;
                    8'd3:  ipv4_total_len[7:0]  <= rx_data;
                    8'd4:  ipv4_id[15:8]        <= rx_data;
                    8'd5:  ipv4_id[7:0]         <= rx_data;
                    8'd6:  begin
                               ipv4_flags          <= rx_data[7:5];
                               ipv4_frag_off[12:8] <= rx_data[4:0];
                           end
                    8'd7:  ipv4_frag_off[7:0]   <= rx_data;
                    8'd8:  ipv4_ttl              <= rx_data;
                    8'd9:  ipv4_protocol         <= rx_data;
                    8'd10: ipv4_checksum[15:8]   <= rx_data;
                    8'd11: ipv4_checksum[7:0]    <= rx_data;
                    8'd12: ipv4_src_addr[31:24]  <= rx_data;
                    8'd13: ipv4_src_addr[23:16]  <= rx_data;
                    8'd14: ipv4_src_addr[15:8]   <= rx_data;
                    8'd15: ipv4_src_addr[7:0]    <= rx_data;
                    8'd16: ipv4_dst_addr[31:24]  <= rx_data;
                    8'd17: ipv4_dst_addr[23:16]  <= rx_data;
                    8'd18: ipv4_dst_addr[15:8]   <= rx_data;
                    8'd19: begin
                               ipv4_dst_addr[7:0] <= rx_data;
                               ipv4_valid         <= 1'b1;
                               ip_version_out     <= 2'd1;
                               parse_done         <= 1'b1;
                           end
                    default: ;
                endcase
            end // pkt_is_ipv4

            // ════════════════════════════════════════════════
            //  IPv6 — 40-byte fixed header
            // ════════════════════════════════════════════════
            if (pkt_is_ipv6) begin
                case (ip_off)
                    8'd0:  begin
                               ipv6_version            <= rx_data[7:4];
                               ipv6_traffic_class[7:4] <= rx_data[3:0];
                               if (rx_data[7:4] != 4'd6)
                                   parse_error <= 1'b1;
                           end
                    8'd1:  begin
                               ipv6_traffic_class[3:0] <= rx_data[7:4];
                               ipv6_flow_label[19:16]  <= rx_data[3:0];
                           end
                    8'd2:  ipv6_flow_label[15:8]  <= rx_data;
                    8'd3:  ipv6_flow_label[7:0]   <= rx_data;
                    8'd4:  ipv6_payload_len[15:8]  <= rx_data;
                    8'd5:  ipv6_payload_len[7:0]   <= rx_data;
                    8'd6:  ipv6_next_header        <= rx_data;
                    8'd7:  ipv6_hop_limit          <= rx_data;
                    // Source address (16 bytes, offsets 8–23)
                    8'd8:  ipv6_src_addr[127:120]  <= rx_data;
                    8'd9:  ipv6_src_addr[119:112]  <= rx_data;
                    8'd10: ipv6_src_addr[111:104]  <= rx_data;
                    8'd11: ipv6_src_addr[103:96]   <= rx_data;
                    8'd12: ipv6_src_addr[95:88]    <= rx_data;
                    8'd13: ipv6_src_addr[87:80]    <= rx_data;
                    8'd14: ipv6_src_addr[79:72]    <= rx_data;
                    8'd15: ipv6_src_addr[71:64]    <= rx_data;
                    8'd16: ipv6_src_addr[63:56]    <= rx_data;
                    8'd17: ipv6_src_addr[55:48]    <= rx_data;
                    8'd18: ipv6_src_addr[47:40]    <= rx_data;
                    8'd19: ipv6_src_addr[39:32]    <= rx_data;
                    8'd20: ipv6_src_addr[31:24]    <= rx_data;
                    8'd21: ipv6_src_addr[23:16]    <= rx_data;
                    8'd22: ipv6_src_addr[15:8]     <= rx_data;
                    8'd23: ipv6_src_addr[7:0]      <= rx_data;
                    // Destination address (16 bytes, offsets 24–39)
                    8'd24: ipv6_dst_addr[127:120]  <= rx_data;
                    8'd25: ipv6_dst_addr[119:112]  <= rx_data;
                    8'd26: ipv6_dst_addr[111:104]  <= rx_data;
                    8'd27: ipv6_dst_addr[103:96]   <= rx_data;
                    8'd28: ipv6_dst_addr[95:88]    <= rx_data;
                    8'd29: ipv6_dst_addr[87:80]    <= rx_data;
                    8'd30: ipv6_dst_addr[79:72]    <= rx_data;
                    8'd31: ipv6_dst_addr[71:64]    <= rx_data;
                    8'd32: ipv6_dst_addr[63:56]    <= rx_data;
                    8'd33: ipv6_dst_addr[55:48]    <= rx_data;
                    8'd34: ipv6_dst_addr[47:40]    <= rx_data;
                    8'd35: ipv6_dst_addr[39:32]    <= rx_data;
                    8'd36: ipv6_dst_addr[31:24]    <= rx_data;
                    8'd37: ipv6_dst_addr[23:16]    <= rx_data;
                    8'd38: ipv6_dst_addr[15:8]     <= rx_data;
                    8'd39: begin
                               ipv6_dst_addr[7:0]  <= rx_data;
                               ipv6_valid          <= 1'b1;
                               ip_version_out      <= 2'd2;
                               parse_done          <= 1'b1;
                           end
                    default: ;
                endcase
            end // pkt_is_ipv6

        end // gcnt >= 14
    end
end

endmodule
